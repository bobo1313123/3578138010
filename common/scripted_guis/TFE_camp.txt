scripted_gui = { #阵营
	##政治界面
	camp_system_politics = { 
		context_type = player_context
        parent_window_token = politics_tab
		window_name = "camp_system_politics"
		ai_enabled = {
			always = no
		}
		properties = {
			camp_button = {
			image = "[GetDynamicFactionIR]"
		}
		}
		
		triggers = {
			camp_button_visible = {
				is_in_faction = yes
			}
		}
		effects = {
            camp_button_click = {
				if = {
					limit = {
						has_country_flag = nato_list_open_politics
					}
					clr_country_flag = nato_list_open_politics
				}
				else = {
					set_country_flag = nato_list_open_politics
				}
				clear_array = global.grid_faction_array_@FROM
				add_to_variable = {
					global.nato_list_variable = 0.01
				}
				FROM = {
					add_to_array = {
						array = global.grid_faction_array_@FROM
					}
				}
				for_each_scope_loop = {
					array = FROM.allies
					THIS = {
					if = {
						limit = {
							PREV = {
								is_faction_leader = yes
							}
						}
						add_to_array = {
							array = global.grid_faction_array_@FROM
							index = 0
						}
					}
					else = {
						add_to_array = {
							array = global.grid_faction_array_@FROM
						}
					}
					}
				}
				
            }
        }
	}
	nato_list_politics = {
        context_type = player_context
        parent_window_token = politics_tab
        window_name = "nato_list_window_politics"
		dirty = nato_list_variable

			always = no
		}

			nato_tags_grid = {

			}
		}
		triggers = {
			major_country_overlay_visible = {

				}
			}
		}

            button_close_click = {

            }
        }
		properties = {
			country_flag = {
				image = "[This.GetFlag]"
			}

			image = "[GetFactionBanner]"
		}
		}
