# 3 路易·贝尔被任命为首相（1935.6.10）
country_event = {
	id = TFE_HOL_EVENT.3
	title = TFE_HOL_EVENT.3.t
	desc = TFE_HOL_EVENT.3.desc
	#picture = GFX_report_event_generic_parliament

	is_triggered_only = yes

	option = {
		name = TFE_HOL_EVENT.3.a

		# 贝尔出任首相，但不取代国家元首：威廉明娜（TVE_HOL_LEADERS，自由主义）继续担任领袖
		recruit_character = HOL_louis_beel
		if = {
			limit = { has_government = Liberalism }
			if = {
				limit = { NOT = { has_character = TVE_HOL_LEADERS } }
				recruit_character = TVE_HOL_LEADERS
			}
			TVE_HOL_LEADERS = { promote_character = liberalism_small }
		}
		add_stability = 0.05
		news_event = { id = TFE_HOL_NEWS.1 days = 1}

		country_event = {
			id = TFE_HOL_EVENT.4
			days = 5
		}
	}
}
